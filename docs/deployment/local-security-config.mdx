# Local Security Configuration Guide

## Overview

This guide provides comprehensive security configuration for Bytebot APIs in local deployments. It covers file-based secrets management, local authentication, SSL/TLS setup, and security hardening—all designed for 100% local operation with no cloud dependencies.

## Security Architecture

### Local Security Model
```
┌─────────────────────────────────────────────────────────────┐
│                    Local Security Stack                     │
├─────────────────────────────────────────────────────────────┤
│  ┌─────────────┐  ┌─────────────┐  ┌─────────────┐         │
│  │    Local    │  │    File     │  │   Local     │         │
```

### Security Principles
- **Zero Trust**: Verify every request and connection
- **Least Privilege**: Minimal permissions for all components
- **Defense in Depth**: Multiple security layers
- **Local Control**: All security controls remain local
- **Audit Trail**: Comprehensive logging of security events

### SSL Certificate Generation (scripts/generate-ssl-certs.sh)
```bash
#!/bin/bash
set -euo pipefail

DOMAIN="localhost"
CERT_DIR="secrets/ssl"
DAYS=365

echo "Generating SSL certificates for local development..."

# Create certificate authority
openssl req -new -x509 -days $DAYS -nodes \
    -out "$CERT_DIR/ca.crt" \
    -keyout "$CERT_DIR/ca.key" \
    -subj "/C=US/ST=Local/L=Development/O=Bytebot/OU=Development/CN=Bytebot-CA"

# Create server private key
openssl genrsa -out "$CERT_DIR/server.key" 2048

# Create certificate signing request
openssl req -new -key "$CERT_DIR/server.key" \
    -out "$CERT_DIR/server.csr" \
    -subj "/C=US/ST=Local/L=Development/O=Bytebot/OU=Development/CN=$DOMAIN"

# Create certificate extensions
cat > "$CERT_DIR/server.ext" << EOF
authorityKeyIdentifier=keyid,issuer
basicConstraints=CA:FALSE
keyUsage = digitalSignature, nonRepudiation, keyEncipherment, dataEncipherment
subjectAltName = @alt_names

[alt_names]
DNS.1 = localhost
DNS.2 = *.localhost
IP.1 = 127.0.0.1
IP.2 = ::1
EOF

# Sign the certificate
openssl x509 -req -in "$CERT_DIR/server.csr" \
    -CA "$CERT_DIR/ca.crt" \
    -CAkey "$CERT_DIR/ca.key" \
    -CAcreateserial \
    -out "$CERT_DIR/server.crt" \
    -days $DAYS \
    -extensions v3_req \
    -extfile "$CERT_DIR/server.ext"

# Generate DH parameters for enhanced security
openssl dhparam -out "$CERT_DIR/dhparam.pem" 2048

# Set proper permissions
chmod 600 "$CERT_DIR/server.key" "$CERT_DIR/ca.key"
chmod 644 "$CERT_DIR/server.crt" "$CERT_DIR/ca.crt"
chmod 644 "$CERT_DIR/dhparam.pem"

# Clean up temporary files
rm "$CERT_DIR/server.csr" "$CERT_DIR/server.ext"

echo "SSL certificates generated successfully!"
echo "Certificate: $CERT_DIR/server.crt"
echo "Private Key: $CERT_DIR/server.key"
echo "CA Certificate: $CERT_DIR/ca.crt"
```

### Secrets Loading Module (src/config/secrets.js)
```javascript
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');

class SecretsManager {
    constructor(secretsPath = './secrets') {
        this.secretsPath = secretsPath;
        this.cache = new Map();
        this.validateSecretsDirectory();
    }

    validateSecretsDirectory() {
        if (!fs.existsSync(this.secretsPath)) {
            throw new Error(`Secrets directory not found: ${this.secretsPath}`);
        }

        // Check directory permissions
        const stats = fs.statSync(this.secretsPath);
        const mode = (stats.mode & parseInt('777', 8)).toString(8);
        if (mode !== '700') {
            console.warn(`Warning: Secrets directory permissions should be 700, current: ${mode}`);
        }
    }

    loadSecret(category, name) {
        const cacheKey = `${category}:${name}`;

        if (this.cache.has(cacheKey)) {
            return this.cache.get(cacheKey);
        }

        const secretPath = path.join(this.secretsPath, category, name);

        if (!fs.existsSync(secretPath)) {
            throw new Error(`Secret not found: ${secretPath}`);
        }

        // Check file permissions
        const stats = fs.statSync(secretPath);
        const mode = (stats.mode & parseInt('777', 8)).toString(8);
        if (mode !== '600' && !name.endsWith('.pem') && !name.endsWith('.crt')) {
            console.warn(`Warning: Secret file permissions should be 600, current: ${mode} for ${secretPath}`);
        }

        const secret = fs.readFileSync(secretPath, 'utf8').trim();
        this.cache.set(cacheKey, secret);

        return secret;
    }

    loadKeyPair(category, privateName, publicName) {
        return {
            private: this.loadSecret(category, privateName),
            public: this.loadSecret(category, publicName)
        };
    }

    loadEnvFile(category, filename) {
        const envPath = path.join(this.secretsPath, category, filename);

        if (!fs.existsSync(envPath)) {
            throw new Error(`Environment file not found: ${envPath}`);
        }

        const envContent = fs.readFileSync(envPath, 'utf8');
        const envVars = {};

        envContent.split('\n').forEach(line => {
            const trimmed = line.trim();
            if (trimmed && !trimmed.startsWith('#')) {
                const [key, ...valueParts] = trimmed.split('=');
                if (key && valueParts.length > 0) {
                    envVars[key.trim()] = valueParts.join('=').trim();
                }
            }
        });

        return envVars;
    }

    generateSecret(length = 32) {
        return crypto.randomBytes(length).toString('base64');
    }

    rotateSecret(category, name) {
        const newSecret = this.generateSecret();
        const secretPath = path.join(this.secretsPath, category, name);

        // Backup existing secret
        if (fs.existsSync(secretPath)) {
            const backupPath = `${secretPath}.backup.${Date.now()}`;
            fs.copyFileSync(secretPath, backupPath);
        }

        // Write new secret
        fs.writeFileSync(secretPath, newSecret, { mode: 0o600 });

        // Update cache
        const cacheKey = `${category}:${name}`;
        this.cache.set(cacheKey, newSecret);

        return newSecret;
    }

    clearCache() {
        this.cache.clear();
    }

    validateSecrets() {
        const requiredSecrets = [
            ['jwt', 'private.pem'],
            ['jwt', 'public.pem'],
            ['jwt', 'refresh.key'],
            ['app', 'session.key'],
            ['app', 'cookie.key'],
            ['app', 'csrf.key'],
            ['ssl', 'server.crt'],
            ['ssl', 'server.key']
        ];

        const missing = [];

        for (const [category, name] of requiredSecrets) {
            const secretPath = path.join(this.secretsPath, category, name);
            if (!fs.existsSync(secretPath)) {
                missing.push(`${category}/${name}`);
            }
        }

        if (missing.length > 0) {
            throw new Error(`Missing required secrets: ${missing.join(', ')}`);
        }

        return true;
    }
}

module.exports = SecretsManager;
```

## Authentication & Authorization

### JWT Authentication Implementation (src/auth/jwt.js)
```javascript
const jwt = require('jsonwebtoken');
const SecretsManager = require('../config/secrets');

class JWTManager {
    constructor() {
        this.secrets = new SecretsManager();
        this.privateKey = this.secrets.loadSecret('jwt', 'private.pem');
        this.publicKey = this.secrets.loadSecret('jwt', 'public.pem');
        this.refreshSecret = this.secrets.loadSecret('jwt', 'refresh.key');

        this.accessTokenExpiry = process.env.JWT_ACCESS_EXPIRY || '15m';
        this.refreshTokenExpiry = process.env.JWT_REFRESH_EXPIRY || '7d';
    }

    generateTokenPair(payload) {
        const accessToken = jwt.sign(
            {
                ...payload,
                type: 'access'
            },
            this.privateKey,
            {
                algorithm: 'RS256',
                expiresIn: this.accessTokenExpiry,
                issuer: 'bytebot-api',
                audience: 'bytebot-client'
            }
        );

        const refreshToken = jwt.sign(
            {
                userId: payload.userId,
                type: 'refresh',
                jti: require('crypto').randomUUID()
            },
            this.refreshSecret,
            {
                algorithm: 'HS256',
                expiresIn: this.refreshTokenExpiry,
                issuer: 'bytebot-api',
                audience: 'bytebot-client'
            }
        );

        return { accessToken, refreshToken };
    }

    verifyAccessToken(token) {
        try {
            return jwt.verify(token, this.publicKey, {
                algorithms: ['RS256'],
                issuer: 'bytebot-api',
                audience: 'bytebot-client'
            });
        } catch (error) {
            throw new Error(`Invalid access token: ${error.message}`);
        }
    }

    verifyRefreshToken(token) {
        try {
            return jwt.verify(token, this.refreshSecret, {
                algorithms: ['HS256'],
                issuer: 'bytebot-api',
                audience: 'bytebot-client'
            });
        } catch (error) {
            throw new Error(`Invalid refresh token: ${error.message}`);
        }
    }

    refreshTokens(refreshToken) {
        const decoded = this.verifyRefreshToken(refreshToken);

        if (decoded.type !== 'refresh') {
            throw new Error('Invalid token type');
        }

        // In production, check if refresh token is blacklisted
        // const isBlacklisted = await this.checkBlacklist(decoded.jti);
        // if (isBlacklisted) throw new Error('Token has been revoked');

        const payload = {
            userId: decoded.userId,
            // Add other necessary claims
        };

        return this.generateTokenPair(payload);
    }

    extractTokenFromHeader(authHeader) {
        if (!authHeader || !authHeader.startsWith('Bearer ')) {
            throw new Error('Invalid authorization header');
        }

        return authHeader.substring(7);
    }
}

module.exports = JWTManager;
```

### Authentication Middleware (src/middleware/auth.js)
```javascript
const JWTManager = require('../auth/jwt');
const rateLimit = require('express-rate-limit');

class AuthMiddleware {
    constructor() {
        this.jwtManager = new JWTManager();
        this.setupRateLimiting();
    }

    setupRateLimiting() {
        this.loginLimiter = rateLimit({
            windowMs: 15 * 60 * 1000, // 15 minutes
            max: 5, // limit each IP to 5 requests per windowMs
            message: 'Too many login attempts, please try again later',
            standardHeaders: true,
            legacyHeaders: false,
            keyGenerator: (req) => {
                return req.ip + ':' + (req.body.email || req.body.username || 'unknown');
            }
        });

        this.apiLimiter = rateLimit({
            windowMs: 15 * 60 * 1000, // 15 minutes
            max: 100, // limit each IP to 100 requests per windowMs
            message: 'Too many API requests, please try again later',
            standardHeaders: true,
            legacyHeaders: false
        });
    }

    authenticate() {
        return async (req, res, next) => {
            try {
                const authHeader = req.headers.authorization;
                const token = this.jwtManager.extractTokenFromHeader(authHeader);
                const decoded = this.jwtManager.verifyAccessToken(token);

                // Verify token type
                if (decoded.type !== 'access') {
                    return res.status(401).json({ error: 'Invalid token type' });
                }

                // Add user info to request
                req.user = {
                    id: decoded.userId,
                    role: decoded.role,
                    permissions: decoded.permissions || []
                };

                // Log authentication event
                this.logAuthEvent(req, 'token_verified', decoded.userId);

                next();
            } catch (error) {
                this.logAuthEvent(req, 'auth_failed', null, error.message);
                res.status(401).json({ error: 'Authentication failed' });
            }
        };
    }

    authorize(requiredRoles = [], requiredPermissions = []) {
        return (req, res, next) => {
            if (!req.user) {
                return res.status(401).json({ error: 'Authentication required' });
            }

            // Check role-based access
            if (requiredRoles.length > 0 && !requiredRoles.includes(req.user.role)) {
                this.logAuthEvent(req, 'access_denied', req.user.id, `Role ${req.user.role} not in ${requiredRoles}`);
                return res.status(403).json({ error: 'Insufficient permissions' });
            }

            // Check permission-based access
            if (requiredPermissions.length > 0) {
                const hasPermission = requiredPermissions.some(permission =>
                    req.user.permissions.includes(permission)
                );

                if (!hasPermission) {
                    this.logAuthEvent(req, 'access_denied', req.user.id, `Missing permissions: ${requiredPermissions}`);
                    return res.status(403).json({ error: 'Insufficient permissions' });
                }
            }

            this.logAuthEvent(req, 'access_granted', req.user.id);
            next();
        };
    }

    csrfProtection() {
        const csrf = require('csurf');
        const SecretsManager = require('../config/secrets');
        const secrets = new SecretsManager();

        return csrf({
            cookie: {
                key: '_csrf',
                secure: process.env.NODE_ENV === 'production',
                httpOnly: true,
                sameSite: 'strict'
            },
            secret: secrets.loadSecret('app', 'csrf.key')
        });
    }

    logAuthEvent(req, event, userId = null, details = null) {
        const logEntry = {
            timestamp: new Date().toISOString(),
            event,
            userId,
            ip: req.ip,
            userAgent: req.get('User-Agent'),
            path: req.path,
            method: req.method,
            details
        };

        // In production, send to secure logging system
        console.log('[AUTH]', JSON.stringify(logEntry));
    }
}

module.exports = AuthMiddleware;
```

```javascript
        const fs = require('fs');
        const path = require('path');
        const sessionsDir = './data/sessions';

        if (!fs.existsSync(sessionsDir)) {
            return;
        }

        const now = Date.now();
        const files = fs.readdirSync(sessionsDir);

        files.forEach(file => {
            const filePath = path.join(sessionsDir, file);
            const stats = fs.statSync(filePath);
            const age = now - stats.mtime.getTime();

            // Remove sessions older than 24 hours
            if (age > 24 * 60 * 60 * 1000) {
                fs.unlinkSync(filePath);
            }
        });
    }
}
```

## API Security

### Input Validation & Sanitization (src/middleware/validation.js)
```javascript
const { body, param, query, validationResult } = require('express-validator');
const DOMPurify = require('isomorphic-dompurify');
const helmet = require('helmet');

class ValidationMiddleware {
    constructor() {
        this.setupHelmet();
    }

    setupHelmet() {
        this.helmetConfig = helmet({
            contentSecurityPolicy: {
                directives: {
                    defaultSrc: ["'self'"],
                    styleSrc: ["'self'", "'unsafe-inline'"],
                    scriptSrc: ["'self'"],
                    imgSrc: ["'self'", "data:", "https:"],
                    connectSrc: ["'self'"],
                    fontSrc: ["'self'"],
                    objectSrc: ["'none'"],
                    mediaSrc: ["'self'"],
                    frameSrc: ["'none'"]
                }
            },
            crossOriginEmbedderPolicy: false,
            hsts: {
                maxAge: 31536000,
                includeSubDomains: true,
                preload: true
            }
        });
    }

    validateRequest() {
        return (req, res, next) => {
            const errors = validationResult(req);
            if (!errors.isEmpty()) {
                return res.status(400).json({
                    error: 'Validation failed',
                    details: errors.array()
                });
            }
            next();
        };
    }

    sanitizeInput() {
        return (req, res, next) => {
            // Sanitize string inputs
            this.sanitizeObject(req.body);
            this.sanitizeObject(req.query);
            this.sanitizeObject(req.params);

            next();
        };
    }

    sanitizeObject(obj) {
        for (const key in obj) {
            if (typeof obj[key] === 'string') {
                obj[key] = DOMPurify.sanitize(obj[key], {
                    ALLOWED_TAGS: [],
                    ALLOWED_ATTR: []
                });
            } else if (typeof obj[key] === 'object' && obj[key] !== null) {
                this.sanitizeObject(obj[key]);
            }
        }
    }

    // Common validation rules
    userValidation = {
        register: [
            body('username')
                .isLength({ min: 3, max: 30 })
                .matches(/^[a-zA-Z0-9_-]+$/)
                .withMessage('Username must be 3-30 characters, alphanumeric, underscore, or hyphen only'),
            body('email')
                .isEmail()
                .normalizeEmail()
                .withMessage('Valid email required'),
            body('password')
                .isLength({ min: 8 })
                .matches(/^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)(?=.*[@$!%*?&])[A-Za-z\d@$!%*?&]/)
                .withMessage('Password must be at least 8 characters with uppercase, lowercase, number, and special character')
        ],
        login: [
            body('email').isEmail().normalizeEmail(),
            body('password').notEmpty()
        ]
    };

    taskValidation = {
        create: [
            body('title')
                .isLength({ min: 1, max: 500 })
                .trim()
                .withMessage('Title must be 1-500 characters'),
            body('description')
                .optional()
                .isLength({ max: 5000 })
                .trim(),
            body('priority')
                .optional()
                .isInt({ min: 1, max: 10 })
                .withMessage('Priority must be between 1 and 10')
        ]
    };

    fileValidation = {
        upload: [
            // File validation handled by multer middleware
            body('description')
                .optional()
                .isLength({ max: 1000 })
                .trim()
        ]
    };
}

module.exports = ValidationMiddleware;
```

### File Upload Security (src/middleware/upload.js)
```javascript
const multer = require('multer');
const path = require('path');
const crypto = require('crypto');
const fs = require('fs');

class SecureFileUpload {
    constructor() {
        this.uploadDir = './data/uploads';
        this.maxFileSize = 10 * 1024 * 1024; // 10MB
        this.allowedMimeTypes = [
            'image/jpeg',
            'image/png',
            'image/gif',
            'application/pdf',
            'text/plain',
            'application/json'
        ];

        this.ensureUploadDir();
        this.setupStorage();
    }

    ensureUploadDir() {
        if (!fs.existsSync(this.uploadDir)) {
            fs.mkdirSync(this.uploadDir, { recursive: true, mode: 0o755 });
        }
    }

    setupStorage() {
        this.storage = multer.diskStorage({
            destination: (req, file, cb) => {
                // Create user-specific subdirectory
                const userDir = path.join(this.uploadDir, String(req.user?.id || 'anonymous'));
                if (!fs.existsSync(userDir)) {
                    fs.mkdirSync(userDir, { recursive: true, mode: 0o755 });
                }
                cb(null, userDir);
            },
            filename: (req, file, cb) => {
                // Generate secure filename
                const ext = path.extname(file.originalname);
                const hash = crypto.randomBytes(16).toString('hex');
                const timestamp = Date.now();
                const filename = `${timestamp}-${hash}${ext}`;
                cb(null, filename);
            }
        });
    }

    fileFilter(req, file, cb) {
        // Check MIME type
        if (!this.allowedMimeTypes.includes(file.mimetype)) {
            return cb(new Error(`File type ${file.mimetype} not allowed`), false);
        }

        // Additional security checks
        const ext = path.extname(file.originalname).toLowerCase();
        const allowedExtensions = ['.jpg', '.jpeg', '.png', '.gif', '.pdf', '.txt', '.json'];

        if (!allowedExtensions.includes(ext)) {
            return cb(new Error(`File extension ${ext} not allowed`), false);
        }

        cb(null, true);
    }

    createUploadMiddleware(fieldName = 'file', maxFiles = 1) {
        const upload = multer({
            storage: this.storage,
            fileFilter: this.fileFilter.bind(this),
            limits: {
                fileSize: this.maxFileSize,
                files: maxFiles
            }
        });

        return (req, res, next) => {
            const uploadHandler = maxFiles === 1 ? upload.single(fieldName) : upload.array(fieldName, maxFiles);

            uploadHandler(req, res, (err) => {
                if (err instanceof multer.MulterError) {
                    if (err.code === 'LIMIT_FILE_SIZE') {
                        return res.status(400).json({ error: 'File too large' });
                    }
                    if (err.code === 'LIMIT_FILE_COUNT') {
                        return res.status(400).json({ error: 'Too many files' });
                    }
                    return res.status(400).json({ error: err.message });
                } else if (err) {
                    return res.status(400).json({ error: err.message });
                }

                // Virus scan placeholder (implement with ClamAV or similar)
                if (req.file || req.files) {
                    this.scanFiles(req.file ? [req.file] : req.files)
                        .then(() => next())
                        .catch(scanErr => {
                            // Clean up uploaded files
                            this.cleanupFiles(req.file ? [req.file] : req.files);
                            res.status(400).json({ error: 'File security scan failed' });
                        });
                } else {
                    next();
                }
            });
        };
    }

    async scanFiles(files) {
        // Implement virus scanning here
        // For now, just basic file validation
        for (const file of files) {
            await this.validateFileContent(file);
        }
    }

    async validateFileContent(file) {
        return new Promise((resolve, reject) => {
            // Basic file header validation
            const buffer = Buffer.alloc(512);
            const fd = fs.openSync(file.path, 'r');

            try {
                fs.readSync(fd, buffer, 0, 512, 0);
                fs.closeSync(fd);

                // Check for common malicious patterns
                const content = buffer.toString('hex');
                const maliciousPatterns = [
                    '4d5a', // PE executable
                    '7f454c46', // ELF executable
                    '213c617263683e' // Archive header
                ];

                for (const pattern of maliciousPatterns) {
                    if (content.includes(pattern)) {
                        reject(new Error('Potentially malicious file detected'));
                        return;
                    }
                }

                resolve();
            } catch (err) {
                fs.closeSync(fd);
                reject(err);
            }
        });
    }

    cleanupFiles(files) {
        files.forEach(file => {
            if (fs.existsSync(file.path)) {
                fs.unlinkSync(file.path);
            }
        });
    }
}
```

```javascript
            max: 30,
            message: 'API rate limit exceeded'
        });

            const clientIP = req.ip || req.connection.remoteAddress;

            }

            next();
        };
    }

        return require('express').json({ limit: maxSize });
```

### Security Audit Logger (src/security/audit.js)
```javascript
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');

class SecurityAuditLogger {
    constructor() {
        this.logDir = './data/logs/security';
        this.ensureLogDir();
        this.setupLogRotation();
    }

    ensureLogDir() {
        if (!fs.existsSync(this.logDir)) {
            fs.mkdirSync(this.logDir, { recursive: true, mode: 0o755 });
        }
    }

    log(event, details = {}) {
        const logEntry = {
            timestamp: new Date().toISOString(),
            event,
            details,
            hash: this.generateHash(event, details)
        };

        const logFile = path.join(this.logDir, `security-${this.getDateString()}.log`);
        const logLine = JSON.stringify(logEntry) + '\n';

        fs.appendFileSync(logFile, logLine, { mode: 0o644 });
    }

    generateHash(event, details) {
        const data = JSON.stringify({ event, details, timestamp: Date.now() });
        return crypto.createHash('sha256').update(data).digest('hex');
    }

    getDateString() {
        return new Date().toISOString().split('T')[0];
    }

    setupLogRotation() {
        // Rotate logs daily at midnight
        const now = new Date();
        const midnight = new Date();
        midnight.setHours(24, 0, 0, 0);

        const msUntilMidnight = midnight.getTime() - now.getTime();

        setTimeout(() => {
            this.rotateLogs();
            // Set up daily rotation
            setInterval(() => this.rotateLogs(), 24 * 60 * 60 * 1000);
        }, msUntilMidnight);
    }

    rotateLogs() {
        const files = fs.readdirSync(this.logDir);
        const thirtyDaysAgo = new Date();
        thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30);

        files.forEach(file => {
            const filePath = path.join(this.logDir, file);
            const stats = fs.statSync(filePath);

            if (stats.birthtime < thirtyDaysAgo) {
                // Archive old logs
                const archiveDir = path.join(this.logDir, 'archive');
                if (!fs.existsSync(archiveDir)) {
                    fs.mkdirSync(archiveDir, { mode: 0o755 });
                }

                const archivePath = path.join(archiveDir, file);
                fs.renameSync(filePath, archivePath);

                // Compress archived logs
                const zlib = require('zlib');
                const gzip = zlib.createGzip();
                const source = fs.createReadStream(archivePath);
                const destination = fs.createWriteStream(`${archivePath}.gz`);

                source.pipe(gzip).pipe(destination);

                destination.on('close', () => {
                    fs.unlinkSync(archivePath);
                });
            }
        });
    }

    // Security event types
    logAuthEvent(userId, event, details = {}) {
        this.log('AUTH_EVENT', { userId, event, ...details });
    }

    logAccessEvent(userId, resource, action, success = true) {
        this.log('ACCESS_EVENT', { userId, resource, action, success });
    }

    logSecurityViolation(type, details = {}) {
        this.log('SECURITY_VIOLATION', { type, ...details });
    }

    logSystemEvent(event, details = {}) {
        this.log('SYSTEM_EVENT', { event, ...details });
    }
}

module.exports = SecurityAuditLogger;
```
