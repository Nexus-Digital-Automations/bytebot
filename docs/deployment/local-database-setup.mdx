# Local Database Setup Guide

## Overview

This guide provides comprehensive instructions for setting up and configuring local databases for Bytebot APIs. We support both SQLite for single-user deployments and PostgreSQL for multi-user environments, ensuring 100% local operation with no cloud dependencies.

## Database Strategy Overview

### Architecture Decision Matrix

| Deployment Type | Database Choice | Use Case | Advantages |
|----------------|----------------|----------|------------|
| **Development** | SQLite | Single developer, testing | Zero setup, file-based, portable |
| **Single-User Production** | SQLite | Personal deployment | Simple, reliable, low maintenance |
| **Multi-User Production** | PostgreSQL | Team/enterprise deployment | ACID compliance, concurrent users, advanced features |
| **High-Performance** | PostgreSQL | Heavy workloads | Advanced indexing, query optimization, scalability |

### SQLite Performance Optimization
```javascript
// Database connection with optimizations (src/database/sqlite.js)
const Database = require('better-sqlite3');
const path = require('path');

class SQLiteDatabase {
    constructor(config) {
        this.config = config;
        this.db = new Database(config.database, {
            verbose: process.env.NODE_ENV === 'development' ? console.log : null
        });

        this.initializeOptimizations();
    }

    initializeOptimizations() {
        // Enable WAL mode for better concurrency
        this.db.pragma('journal_mode = WAL');

        // Optimize for performance
        this.db.pragma('synchronous = NORMAL');
        this.db.pragma('cache_size = -64000'); // 64MB cache
        this.db.pragma('temp_store = MEMORY');
        this.db.pragma('mmap_size = 268435456'); // 256MB mmap

        // Enable foreign keys
        this.db.pragma('foreign_keys = ON');

        // Analyze tables for better query planning
        this.db.pragma('analysis_limit = 1000');
        this.db.pragma('optimize');
    }

    backup(backupPath) {
        const backup = this.db.backup(backupPath);
        return new Promise((resolve, reject) => {
            backup.step(-1, (remaining, pageCount) => {
                if (remaining === 0) {
                    backup.finish();
                    resolve({ pageCount });
                }
            });
        });
    }

    vacuum() {
        this.db.exec('VACUUM;');
    }

    close() {
        this.db.close();
    }
}

module.exports = SQLiteDatabase;
```

## Database Connection Management

### Connection Pool Configuration (src/database/pool.js)
```javascript
const { Pool } = require('pg');
const Database = require('better-sqlite3');

class DatabaseManager {
    constructor(config) {
        this.config = config;
        this.type = config.type; // 'sqlite' or 'postgresql'
        this.pool = null;
        this.db = null;

        this.initialize();
    }

    initialize() {
        if (this.type === 'postgresql') {
            this.pool = new Pool({
                host: this.config.postgresql.host,
                port: this.config.postgresql.port,
                database: this.config.postgresql.database,
                user: this.config.postgresql.username,
                password: this.config.postgresql.password,
                ssl: this.config.postgresql.ssl,
                ...this.config.postgresql.pool
            });

            // Connection error handling
            this.pool.on('error', (err) => {
                console.error('Unexpected error on idle client', err);
            });

        } else if (this.type === 'sqlite') {
            this.db = new Database(this.config.sqlite.database, {
                verbose: process.env.NODE_ENV === 'development' ? console.log : null
            });

            // Apply SQLite optimizations
            Object.entries(this.config.sqlite.options).forEach(([key, value]) => {
                this.db.pragma(`${key} = ${value}`);
            });
        }
    }

    async query(text, params = []) {
        if (this.type === 'postgresql') {
            const client = await this.pool.connect();
            try {
                const result = await client.query(text, params);
                return result;
            } finally {
                client.release();
            }
        } else if (this.type === 'sqlite') {
            const stmt = this.db.prepare(text);
            return stmt.all(params);
        }
    }

    async transaction(callback) {
        if (this.type === 'postgresql') {
            const client = await this.pool.connect();
            try {
                await client.query('BEGIN');
                const result = await callback(client);
                await client.query('COMMIT');
                return result;
            } catch (error) {
                await client.query('ROLLBACK');
                throw error;
            } finally {
                client.release();
            }
        } else if (this.type === 'sqlite') {
            return this.db.transaction(callback)();
        }
    }

    async close() {
        if (this.pool) {
            await this.pool.end();
        }
        if (this.db) {
            this.db.close();
        }
    }

    async healthCheck() {
        try {
            if (this.type === 'postgresql') {
                const result = await this.query('SELECT NOW() as current_time');
                return { status: 'healthy', timestamp: result.rows[0].current_time };
            } else if (this.type === 'sqlite') {
                const result = this.query("SELECT datetime('now') as current_time");
                return { status: 'healthy', timestamp: result[0].current_time };
            }
        } catch (error) {
            return { status: 'unhealthy', error: error.message };
        }
    }
}

module.exports = DatabaseManager;
```

## Backup and Recovery

### Automated Backup Scripts

#### SQLite Backup (scripts/backup-sqlite.sh)
```bash
#!/bin/bash
set -euo pipefail

# Configuration
BACKUP_DIR="./data/backups/sqlite"
DB_FILE="./data/sqlite/bytebot.db"
RETENTION_DAYS=7

# Create backup directory
mkdir -p "$BACKUP_DIR"

# Generate backup filename with timestamp
BACKUP_FILE="$BACKUP_DIR/bytebot-$(date +%Y%m%d_%H%M%S).db"

# Create backup using SQLite backup API
sqlite3 "$DB_FILE" ".backup $BACKUP_FILE"

# Compress backup
gzip "$BACKUP_FILE"

# Verify backup integrity
if gunzip -t "$BACKUP_FILE.gz"; then
    echo "Backup created successfully: $BACKUP_FILE.gz"
else
    echo "Backup verification failed!" >&2
    exit 1
fi

# Clean up old backups
find "$BACKUP_DIR" -name "*.db.gz" -mtime +$RETENTION_DAYS -delete

echo "SQLite backup completed: $BACKUP_FILE.gz"
```

#### PostgreSQL Backup (scripts/backup-postgresql.sh)
```bash
#!/bin/bash
set -euo pipefail

# Configuration
BACKUP_DIR="./data/backups/postgresql"
DB_NAME="bytebot"
DB_USER="bytebot"
RETENTION_DAYS=30

# Create backup directory
mkdir -p "$BACKUP_DIR"

# Generate backup filename with timestamp
BACKUP_FILE="$BACKUP_DIR/bytebot-$(date +%Y%m%d_%H%M%S).sql"

# Create backup using pg_dump
pg_dump -h localhost -U "$DB_USER" -d "$DB_NAME" \
    --verbose --clean --if-exists --no-owner --no-privileges \
    -f "$BACKUP_FILE"

# Compress backup
gzip "$BACKUP_FILE"

# Verify backup integrity
if gunzip -t "$BACKUP_FILE.gz"; then
    echo "Backup created successfully: $BACKUP_FILE.gz"
else
    echo "Backup verification failed!" >&2
    exit 1
fi

# Clean up old backups
find "$BACKUP_DIR" -name "*.sql.gz" -mtime +$RETENTION_DAYS -delete

echo "PostgreSQL backup completed: $BACKUP_FILE.gz"
```

### Recovery Procedures

#### SQLite Recovery
```bash
# 1. Stop application
sudo systemctl stop bytebot-api

# 2. Backup current database (if corrupted)
cp data/sqlite/bytebot.db data/sqlite/bytebot.db.corrupted

# 3. Restore from backup
gunzip -c data/backups/sqlite/bytebot-YYYYMMDD_HHMMSS.db.gz > data/sqlite/bytebot.db

# 4. Verify integrity
sqlite3 data/sqlite/bytebot.db "PRAGMA integrity_check;"

# 5. Start application
sudo systemctl start bytebot-api
```

#### PostgreSQL Recovery
```bash
# 1. Stop application
sudo systemctl stop bytebot-api

# 2. Drop and recreate database
sudo -u postgres psql -c "DROP DATABASE IF EXISTS bytebot;"
sudo -u postgres psql -c "CREATE DATABASE bytebot OWNER bytebot;"

# 3. Restore from backup
gunzip -c data/backups/postgresql/bytebot-YYYYMMDD_HHMMSS.sql.gz | \
    psql -h localhost -U bytebot -d bytebot

# 4. Verify restoration
psql -h localhost -U bytebot -d bytebot -c "SELECT COUNT(*) FROM users;"

# 5. Start application
sudo systemctl start bytebot-api
```

```javascript
            memory: process.memoryUsage(),
        };

        if (req.db.type === 'postgresql') {
            const poolStats = {
                totalCount: req.db.pool.totalCount,
                idleCount: req.db.pool.idleCount,
                waitingCount: req.db.pool.waitingCount
            };
            stats.pool = poolStats;
        }

        res.json({
            status: 'healthy',
            database: stats
        });

    } catch (error) {
        res.status(500).json({
            status: 'unhealthy',
            error: error.message
        });
    }
});

module.exports = router;
```
