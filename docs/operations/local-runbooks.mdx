# Local Deployment Operational Runbooks

## Overview

This comprehensive runbook provides step-by-step operational procedures for managing Bytebot APIs in local deployments. All procedures are designed for 100% local operation with no cloud dependencies, ensuring complete operational control.

## Daily Operations

### Morning Health Check Routine

#### System Status Verification
```bash
#!/bin/bash
# daily-health-check.sh

echo "=== Bytebot Local Deployment Health Check ==="
echo "Date: $(date)"
echo ""

# 1. Check system resources
echo "1. System Resources:"
echo "   CPU Usage: $(top -bn1 | grep "Cpu(s)" | awk '{print $2}' | cut -d'%' -f1)%"
echo "   Memory: $(free -h | awk 'NR==2{printf "%.1f%%", $3*100/$2}')"
echo "   Disk: $(df -h / | awk 'NR==2{print $5}')"
echo ""

# 2. Check service status
echo "2. Service Status:"
services=("bytebot-api" "postgres" "redis" "prometheus" "grafana")
for service in "${services[@]}"; do
    if systemctl is-active --quiet $service; then
        echo "   ✓ $service: Running"
    else
        echo "   ✗ $service: Not running"
    fi
done
echo ""

# 3. Check API health
echo "3. API Health:"
api_health=$(curl -s http://localhost:3000/api/v1/health | jq -r '.data.status' 2>/dev/null || echo "unreachable")
echo "   API Status: $api_health"
echo ""

# 4. Check database connectivity
echo "4. Database Status:"
if psql -h localhost -U bytebot -d bytebot -c "SELECT 1;" >/dev/null 2>&1; then
    echo "   ✓ PostgreSQL: Connected"
else
    echo "   ✗ PostgreSQL: Connection failed"
fi
echo ""

# 5. Check disk space for critical paths
echo "5. Storage Status:"
critical_paths=("/data" "/data/logs" "/data/backups")
for path in "${critical_paths[@]}"; do
    if [ -d "$path" ]; then
        usage=$(df -h "$path" | awk 'NR==2{print $5}' | cut -d'%' -f1)
        if [ "$usage" -gt 80 ]; then
            echo "   ⚠ $path: ${usage}% (Warning: >80%)"
        else
            echo "   ✓ $path: ${usage}%"
        fi
    else
        echo "   ✗ $path: Directory not found"
    fi
done
echo ""

# 6. Check log errors from last 24 hours
echo "6. Recent Errors:"
error_count=$(grep -c "ERROR" /data/logs/application/app.log 2>/dev/null || echo "0")
echo "   Errors in last 24h: $error_count"
echo ""

echo "=== Health Check Complete ==="
```

### Evening Backup Routine

#### Automated Backup Script
```bash
#!/bin/bash
# evening-backup.sh

set -euo pipefail

BACKUP_DIR="/data/backups/$(date +%Y%m%d)"
LOG_FILE="/data/logs/backup.log"

echo "$(date): Starting evening backup" >> "$LOG_FILE"

# Create backup directory
mkdir -p "$BACKUP_DIR"

# 1. Database backup
echo "$(date): Backing up database" >> "$LOG_FILE"
pg_dump -h localhost -U bytebot -d bytebot | gzip > "$BACKUP_DIR/postgres.sql.gz"

# 2. Application data backup
echo "$(date): Backing up application data" >> "$LOG_FILE"
tar -czf "$BACKUP_DIR/app-data.tar.gz" \
    --exclude="*.log" \
    --exclude="*.tmp" \
    /data/uploads /data/agents /data/sessions

# 3. Configuration backup
echo "$(date): Backing up configuration" >> "$LOG_FILE"
tar -czf "$BACKUP_DIR/config.tar.gz" \
    config/ secrets/ docker-compose.yml .env

# 4. Verify backup integrity
echo "$(date): Verifying backup integrity" >> "$LOG_FILE"
for file in "$BACKUP_DIR"/*.gz; do
    if ! gunzip -t "$file" 2>/dev/null; then
        echo "$(date): ERROR: Backup verification failed for $file" >> "$LOG_FILE"
        exit 1
    fi
done

# 5. Clean old backups (keep 7 days)
find /data/backups -type d -name "????????" -mtime +7 -exec rm -rf {} \;

# 6. Update latest symlink
ln -sfn "$BACKUP_DIR" /data/backups/latest

echo "$(date): Backup completed successfully" >> "$LOG_FILE"
```

## Incident Response Procedures

### Service Down Response

#### API Service Down
```bash
#!/bin/bash
# incident-api-down.sh

echo "=== API SERVICE DOWN INCIDENT RESPONSE ==="

# 1. Immediate assessment
echo "1. Checking API status..."
API_STATUS=$(curl -s -w "%{http_code}" -o /dev/null http://localhost:3000/health || echo "000")
echo "   HTTP Status: $API_STATUS"

if [ "$API_STATUS" != "200" ]; then
    echo "2. Checking API process..."
    if pgrep -f "node.*api" >/dev/null; then
        echo "   Process running but not responding"
        echo "   Collecting logs..."
        tail -100 /data/logs/application/app.log > "/tmp/api-incident-$(date +%s).log"

        echo "   Attempting graceful restart..."
        pkill -TERM -f "node.*api"
        sleep 10
    else
        echo "   Process not running"
    fi

    echo "3. Starting API service..."
    systemctl start bytebot-api

    echo "4. Waiting for recovery..."
    for i in {1..30}; do
        sleep 2
        if curl -s http://localhost:3000/health >/dev/null 2>&1; then
            echo "   ✓ API service recovered"
            break
        fi
        echo "   Waiting... ($i/30)"
    done
fi

echo "5. Final status check..."
curl -s http://localhost:3000/api/v1/health | jq .
```

        echo "   ✓ Database recovered after restart"

### Security Maintenance

#### Security Audit Procedures
```bash
#!/bin/bash
# security-audit.sh

echo "=== SECURITY AUDIT PROCEDURES ==="

# 1. Check file permissions
echo "1. Checking critical file permissions..."
declare -A expected_perms=(
    ["/data"]="755"
    ["/data/uploads"]="755"
    ["/data/logs"]="755"
    ["secrets/"]="700"
    ["secrets/jwt/private.pem"]="600"
    ["secrets/ssl/server.key"]="600"
)

for path in "${!expected_perms[@]}"; do
    if [ -e "$path" ]; then
        actual_perm=$(stat -c "%a" "$path")
        expected_perm="${expected_perms[$path]}"
        if [ "$actual_perm" = "$expected_perm" ]; then
            echo "   ✓ $path: $actual_perm"
        else
            echo "   ⚠ $path: $actual_perm (expected: $expected_perm)"
        fi
    else
        echo "   ✗ $path: Not found"
    fi
done

# 2. Check for failed authentication attempts
echo "2. Checking authentication logs..."
FAILED_ATTEMPTS=$(grep -c "auth_failed" /data/logs/security/security.log 2>/dev/null || echo "0")
echo "   Failed authentication attempts in last 24h: $FAILED_ATTEMPTS"

# 3. Check for security violations
echo "3. Checking security violations..."
VIOLATIONS=$(grep -c "security_violation" /data/logs/security/security.log 2>/dev/null || echo "0")
echo "   Security violations in last 24h: $VIOLATIONS"

# 4. Check SSL certificate expiry
echo "4. Checking SSL certificate..."
if [ -f "secrets/ssl/server.crt" ]; then
    CERT_EXPIRY=$(openssl x509 -in secrets/ssl/server.crt -noout -enddate | cut -d= -f2)
    CERT_EXPIRY_EPOCH=$(date -d "$CERT_EXPIRY" +%s)
    CURRENT_EPOCH=$(date +%s)
    DAYS_TO_EXPIRY=$(( (CERT_EXPIRY_EPOCH - CURRENT_EPOCH) / 86400 ))

    if [ $DAYS_TO_EXPIRY -lt 30 ]; then
        echo "   ⚠ SSL certificate expires in $DAYS_TO_EXPIRY days"
    else
        echo "   ✓ SSL certificate valid for $DAYS_TO_EXPIRY days"
    fi
fi

# 5. Check for weak passwords (simplified check)
echo "5. Checking password policies..."
WEAK_PASSWORDS=$(psql -h localhost -U bytebot -d bytebot -t -c "
SELECT COUNT(*) FROM users
WHERE LENGTH(password_hash) < 50;")
echo "   Users with potentially weak passwords: $WEAK_PASSWORDS"

echo "Security audit complete"
```

## Recovery Procedures

### Disaster Recovery

#### Complete System Recovery
```bash
#!/bin/bash
# disaster-recovery.sh

set -euo pipefail

echo "=== DISASTER RECOVERY PROCEDURE ==="

BACKUP_SOURCE="/data/backups/latest"
RECOVERY_LOG="/tmp/recovery-$(date +%s).log"

exec 1> >(tee -a "$RECOVERY_LOG")
exec 2>&1

# 1. Verify backup integrity
echo "1. Verifying backup integrity..."
if [ ! -d "$BACKUP_SOURCE" ]; then
    echo "ERROR: Backup source not found: $BACKUP_SOURCE"
    exit 1
fi

for backup_file in "$BACKUP_SOURCE"/*.gz; do
    if ! gunzip -t "$backup_file" 2>/dev/null; then
        echo "ERROR: Corrupted backup file: $backup_file"
        exit 1
    fi
done
echo "   ✓ All backup files verified"

# 2. Stop all services
echo "2. Stopping all services..."
systemctl stop bytebot-api postgresql redis-server grafana prometheus

# 3. Restore database
echo "3. Restoring database..."
# Drop and recreate database
sudo -u postgres psql -c "DROP DATABASE IF EXISTS bytebot;"
sudo -u postgres psql -c "CREATE DATABASE bytebot OWNER bytebot;"

# Restore from backup
gunzip -c "$BACKUP_SOURCE/postgres.sql.gz" | psql -h localhost -U bytebot -d bytebot
echo "   ✓ Database restored"

# 4. Restore application data
echo "4. Restoring application data..."
rm -rf /data/uploads /data/agents /data/sessions
tar -xzf "$BACKUP_SOURCE/app-data.tar.gz" -C /
echo "   ✓ Application data restored"

# 5. Restore configuration
echo "5. Restoring configuration..."
tar -xzf "$BACKUP_SOURCE/config.tar.gz" -C /
echo "   ✓ Configuration restored"

# 6. Set proper permissions
echo "6. Setting permissions..."
chown -R bytebot:bytebot /data
chmod -R 755 /data
chmod -R 700 secrets/
chmod 600 secrets/*/*.key secrets/*/*.pem
echo "   ✓ Permissions set"

# 7. Start services
echo "7. Starting services..."
systemctl start postgresql redis-server
sleep 10
systemctl start bytebot-api grafana prometheus

# 8. Verify recovery
echo "8. Verifying recovery..."
sleep 30

# Check database
if psql -h localhost -U bytebot -d bytebot -c "SELECT COUNT(*) FROM users;" >/dev/null 2>&1; then
    echo "   ✓ Database functional"
else
    echo "   ✗ Database check failed"
fi

# Check API
if curl -s http://localhost:3000/health >/dev/null 2>&1; then
    echo "   ✓ API functional"
else
    echo "   ✗ API check failed"
fi

echo "=== DISASTER RECOVERY COMPLETE ==="
echo "Recovery log saved to: $RECOVERY_LOG"
```

    "load_average": [$(uptime | awk -F'load average:' '{print $2}' | sed 's/,//g')]
  },
  "database": {
    "connections": $(psql -h localhost -U bytebot -d bytebot -t -c "SELECT count(*) FROM pg_stat_activity;" 2>/dev/null | tr -d ' ' || echo "0"),
    "size_mb": $(psql -h localhost -U bytebot -d bytebot -t -c "SELECT pg_size_pretty(pg_database_size('bytebot'));" 2>/dev/null | tr -d ' ' || echo "0")
  }
}
EOF
)

echo "$SYSTEM_METRICS" > "$BASELINE_FILE"
echo "Performance baseline saved to: $BASELINE_FILE"
